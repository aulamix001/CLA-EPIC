cmake_minimum_required(VERSION 3.22)
project(EpicLab VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EPIC_DSP_ONLY "Build portable DSP tests without JUCE" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(dsp_tests tests/dsp_tests.cpp)
  target_include_directories(dsp_tests PRIVATE Source)
  add_test(NAME dsp_tests COMMAND dsp_tests)
  add_executable(preset_tests tests/preset_tests.cpp)
  target_include_directories(preset_tests PRIVATE Source)
  add_test(NAME preset_tests COMMAND preset_tests)
endif()
if(EPIC_DSP_ONLY)
  return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_binary_data(EpicAssets SOURCES assets/imagem.png)
juce_add_plugin(EpicLab COMPANY_NAME "EpicLab" PLUGIN_MANUFACTURER_CODE EpLb PLUGIN_CODE Ep01
  FORMATS VST3 Standalone PRODUCT_NAME "EpicLab" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
  NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Delay Reverb)
target_sources(EpicLab PRIVATE Source/Plugin.cpp Source/Dsp.h Source/Presets.h Source/Editor.h)
target_link_libraries(EpicLab PRIVATE EpicAssets juce::juce_audio_utils juce::juce_recommended_config_flags)
target_compile_definitions(EpicLab PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
if(BUILD_TESTING)
  juce_add_console_app(interface_tests PRODUCT_NAME "EpicLab Interface Tests")
  target_sources(interface_tests PRIVATE tests/interface_tests.cpp)
  target_include_directories(interface_tests PRIVATE Source)
  target_link_libraries(interface_tests PRIVATE EpicLab juce::juce_audio_utils)
  add_test(NAME interface_tests COMMAND interface_tests)
endif()
